cmake_minimum_required(VERSION 3.16)
project(forcevla_offline_ik LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(pinocchio REQUIRED)
find_package(yaml-cpp REQUIRED)
file(GLOB DAQP_SOURCES CONFIGURE_DEPENDS "vendor/daqp-0.7.2/src/*.c")
add_library(forcevla_daqp STATIC ${DAQP_SOURCES})
target_include_directories(forcevla_daqp PUBLIC vendor/daqp-0.7.2/include)
set(QP_PREFIX "/opt/kernelmind/apex/install/marvin_qp_controller" CACHE PATH "Installed QP core prefix")
add_executable(offline_ik_worker worker.cpp)
target_include_directories(offline_ik_worker PRIVATE "${QP_PREFIX}/include" vendor)
target_link_libraries(offline_ik_worker PRIVATE "${QP_PREFIX}/lib/marvin_qp_controller/libqp_controller_core.a" forcevla_daqp pinocchio::pinocchio yaml-cpp)
