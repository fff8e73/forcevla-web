cmake_minimum_required(VERSION 3.16)
project(forcevla_loop_motion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(marvin_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(loop_motion loop_motion.cpp)
target_include_directories(loop_motion PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
ament_target_dependencies(loop_motion rclcpp marvin_msgs std_msgs)
target_link_libraries(loop_motion pthread dl)
